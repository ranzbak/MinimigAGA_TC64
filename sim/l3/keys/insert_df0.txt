# sim/l3/keys/insert_df0.txt: the key script of scenario fdd_rd (bfm_keys.v,
# syntax there; README "keys" and "fdd_rd").  The fdd card's menu
# (fw/ctrl_rv/menu.c), from the main-loop milestone: open the OSD, turn the
# floppy drive sounds on (the card's config has drivesounds 0; DRIVESND.BIN
# is loaded, so the Floppy page's "Floppy disk sounds" line is selectable),
# insert TEST.ADF into df0 through its file selector, close the OSD.
# l3check.py fdd_rd reads the pages back: "Floppy disk sounds : on" and
# "df0: TEST".
#
# Each key as in menu_tour.txt: WAIT_KEYQ 1 (applied right after a KEYQ
# frame, so the next KEYQ frame drains it and no level read sees it),
# KEY <code> 64, WAIT_KEYQ 1 (that frame), WAIT_OSD_IDLE 100000 (the page
# drawn).  Codes (fw/ctrl_rv/osd.h): 69 KEY_MENU, 4c KEY_UP, 4d KEY_DOWN,
# 44 KEY_ENTER.  menusub = the selected line (menu.c).
# open the OSD (MENU_NONE2 -> MENU_MAIN1; main page, line 0 = df0)
WAIT_KEYQ 1
KEY 69 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000

# --- the Floppy page: drive sounds on (menumask 0xf3 on the main page: df0, df1, Floppy, Hard disk, CD-ROM, exit)
# down: df1
WAIT_KEYQ 1
KEY 4d 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# down: Floppy disk settings
WAIT_KEYQ 1
KEY 4d 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# enter: the Floppy page (MENU_FLOPPY1), line 0, parentsub 4
WAIT_KEYQ 1
KEY 44 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# down: Floppy disk turbo
WAIT_KEYQ 1
KEY 4d 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# down: Floppy disk sounds (selectable: drivesounds_loaded(), menumask 0x0f)
WAIT_KEYQ 1
KEY 4d 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# enter: config.drivesounds ^= DRIVESOUNDS_FLOPPY, drivesounds_enable; the page again, "on"
WAIT_KEYQ 1
KEY 44 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# down: exit
WAIT_KEYQ 1
KEY 4d 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# enter on exit: the main page, menusub = parentsub = 4 (Floppy disk settings)
WAIT_KEYQ 1
KEY 44 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000

# --- df0: the file selector, TEST.ADF
# up: df1
WAIT_KEYQ 1
KEY 4c 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# up: df0
WAIT_KEYQ 1
KEY 4c 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# enter on df0 (no disk): SelectFile("ADF"), the file selector (MENU_FILE_SELECT1); TEST.ADF is the card's only ADF, selected
WAIT_KEYQ 1
KEY 44 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
# enter: MENU_FILE_SELECTED, InsertFloppy(&df[0]), the main page with "df0: TEST", menusub 1
WAIT_KEYQ 1
KEY 44 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000

# --- close the OSD (MENU_MAIN2: menu -> MENU_NONE1, OsdHide)
WAIT_KEYQ 1
KEY 69 64
WAIT_KEYQ 1
WAIT_OSD_IDLE 100000
